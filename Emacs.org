* Conventions
Let's try and keep a readable, maintainable, modern, consistent style.
 * We assume NixOS installs every needed package (=common/apps/emacs.nix=), so we don't have any =:ensure=.
 * Make 1 use-package block per library, and the name is determined by the options being set (do C-h v on an option).
   * That lets us use =nij/audit-custom=.
 * Keyword order:
   preface, after/defer/demand, init, hook, mode, bind, commands, custom, custom-face then config.
 * Helper functions use the nij/ namespace, and live in the preface of their own block rather than being gathered together
 * Global minor-modes - do =:init (foo-mode 1)= so that it runs even if the rest of the block is lazy.
 * =defcustom= options go in :custom. Plain variables are =:config (setq foo 42)=. :custom values don't call the package's own functions
 * hook, mode, bind and commands all make it lazy
 * :defer t for when there are only settings to be loaded by another block
 * Prefix keys - defvar-keymap maps.

** Core Emacs Config
This section configures core Emacs behavior, fonts, and helper functions. Start by setting up ways to look for errors.
*** Find silent errors
=:custom= fails silently: a misspelt name, or a variable that isn't a user option, is recorded and never applied. =M-x nij/audit-custom= checks every variable set through =:custom= in this file. It loads each block's library first, so it takes a moment the first time. An empty list is the result to aim for.
#+begin_src emacs-lisp
  (defun nij/audit-custom ()
    "Report variables set with use-package `:custom' that are not user options.
  Each block's library is loaded first, so anything still listed is a
  typo, a plain variable that wants `setq' in `:config', or an option
  filed under a block that isn't the library defining it."
    (interactive)
    (let (suspects)
      (pcase-dolist (`(,prop ,symbol . ,_) (get 'use-package 'theme-settings))
        (when (eq prop 'theme-value)
          ;; use-package records which block set the variable
          (let ((comment (get symbol 'saved-variable-comment)))
            (when (and (stringp comment)
                       (string-match "use-package \\(.+\\)\\'" comment))
              (ignore-errors (require (intern (match-string 1 comment))))))
          (unless (custom-variable-p symbol)
            (push symbol suspects))))
      (message "Not user options: %S" suspects)))
#+end_src
*** No Littering
By default, packages dump various files and dirs into =.emacs.d/= or =.config/emacs/= - let's make them use ./var and ./etc
=no-littering-theme-backups= centralises backup and autosave files to =var/backup/=. However, for /sudo::/path/to/file, this would copy root-owned files to my $HOME, probably not what I want.
#+begin_src emacs-lisp
  (use-package no-littering
  :preface
  (defun nij/privileged-file-p (name)
    "Return non-nil if NAME is a file opened through TRAMP's su, sudo or doas.
For a multi-hop name such as /ssh:host|sudo:host:/etc/x, the last hop
decides."
    (member (file-remote-p name 'method) '("su" "sudo" "doas")))
  (defun nij/backup-enable-p (name)
    "Like `normal-backup-enable-predicate', but never back up a privileged file."
    (and (normal-backup-enable-predicate name)
         (not (nij/privileged-file-p name))))
  (defun nij/no-auto-save-for-privileged-file ()
    "Turn off auto-saving if this buffer visits a privileged file."
    (when (nij/privileged-file-p buffer-file-name)
      (auto-save-mode -1)))
  :config
  (no-littering-theme-backups)
  (setq backup-enable-predicate #'nij/backup-enable-p)
  ;; add-hook, not :hook: :hook would make this whole block lazy
  (add-hook 'find-file-hook #'nij/no-auto-save-for-privileged-file))
#+end_src

*** Running commands in a compile buffer
Several of my commands use a buffer created by =compile= to run a shell command
=compile= likes to change its global default to whatever's just been run, and this is not safe.
Otherwise a nixos switch or a cmake configure becomes the global default and can be run inappropriately.
#+begin_src emacs-lisp
  (defun nij/compile (command &optional comint)
    "Run COMMAND like `compile', without changing `compile-command'.
  The command runs in `default-directory'. With a non-nil COMINT,
  the buffer accepts input.
  Return the compilation buffer."
    (let ((compile-command compile-command)) (compile command comint)))

  (use-package ansi-color
    :hook (compilation-filter . ansi-color-compilation-filter))

  (defun nij/restart-emacs ()
    "Restart Emacs, first asking about anything that would be lost.
  As a daemon, restart through systemd.  Otherwise use `restart-emacs',
  which asks the same questions itself."
    (interactive)
    (if (not (daemonp))

      (unless (run-hook-with-args-until-failure 'kill-emacs-query-functions)
        (user-error "Restart cancelled"))
      (let ((default-directory "/"))
        (call-process "systemctl" nil 0 nil "--user" "--no-block" "restart" "emacs.service"))))

*** Jump to Last Change
Identify where in a buffer I was most recently editing, using the =buffer-undo-list=
=(BEG . END)= records an insertion.
=(TEXT . POSITION)= records a deletion.
~C-u C-SPC~ goes back to where point was.
#+begin_src emacs-lisp
  (defun nij/goto-last-change ()
    "Move point to the most recent change in this buffer.
    The mark is left where point was."
    (interactive)
    (let ((pos (and (listp buffer-undo-list)   ; t means undo is off in this buffer
                    (seq-some
                     (lambda (entry)
                       (pcase entry
                         ;; insertion: (BEG . END)
                         (`(,(pred integerp) . ,(and (pred integerp) end)) end)
                         ;; deletion: (TEXT . POSITION), negative if point was at the end
                         (`(,(pred stringp) . ,(and (pred integerp) pos)) (abs pos))))
                     buffer-undo-list))))
      (unless pos
        (user-error "No change recorded in this buffer"))
      (push-mark)
      (goto-char pos)))

  (keymap-set goto-map "l" #'nij/goto-last-change) ; M-g l
#+end_src

** Theming, Completion and Help
*** Window Layout
I like a frame to have one main window to the left, and a column of small windows on the right - a bit like a WM master-stack layout.
The right-hand column is a good place for help-type buffers and compile-type buffers.
We can mostly control this with built-ins:
=C-x w s= hide/unhide the whole right-hand column
=M-o= or =C-x o= move to next window
=C-x O= move to prev window
=C-c <left>= and =C-c <right>= to undo and redo a window layout.
Also worth looking at =C-x w t=, as well as the =C-x w o=, =C-x w r= and =C-x w f= maps.
#+begin_src emacs-lisp
  (use-package window

    :custom

            (derived-mode . compilation-mode))
        (display-buffer-reuse-window display-buffer-in-side-window); the 2 possible actions

*** Basic Config
Fonts, per-buffer display, and the babel languages. conf-unix does key-value pairs (eg life=42). More info [[https://orgmode.org/worg/org-contrib/babel/languages.html][here]].

  (defun nij/create-cpp-project ()
    "Scaffold a C++23 project (CMake, Ninja, Nix dev shell) in this directory.
    Every file in `nij/cpp-template-directory' is copied here, with
    {{project}} replaced by the name of this directory.  A template called
    dot-NAME becomes .NAME.  Files that already exist are left alone."
    (interactive)
    (unless (file-directory-p nij/cpp-template-directory)
      (user-error "No templates in %s; save Emacs.org to tangle them"
                  nij/cpp-template-directory))
    (let ((name (file-name-nondirectory (directory-file-name default-directory)))
          files)
      (dolist (template (directory-files nij/cpp-template-directory t
                                         directory-files-no-dot-files-regexp))
        (let ((file (replace-regexp-in-string
                     "\\`dot-" "." (file-name-nondirectory template))))
          (push file files)
          (unless (file-exists-p file)
            (with-temp-file (expand-file-name file)
              (insert-file-contents template)
              (while (search-forward "{{project}}" nil t)
                (replace-match name t t))))))
      ;; Inside a git repository, a flake only sees the files git knows about
      (when (locate-dominating-file default-directory ".git")
        (unless (eql 0 (apply #'process-file "git" nil nil nil "add" "--" files))
          (message "git add failed; the flake may not see the new files")))
      (message "C++ project '%s' scaffolded." name)
      ;; Build the dev shell in a compilation buffer so we can watch, then
      ;; pick up the new environment and open main.cpp
      (with-current-buffer (nij/compile "direnv allow && direnv exec . true")
        (add-hook 'compilation-finish-functions
                  (lambda (_buffer status)
                    (when (string-prefix-p "finished" status)
                      (envrc-reload)
                      (find-file "main.cpp")))
                  nil t))))             ; t: only for this compilation buffer

                   ;; Thror (Ubuntu NAS) - Standard User
                   ((member target '("thror" "t")) "/ssh:thror:/home/nij")
                   
                   ;; Thror - Root Access (TRAMP Multi-hop)
                   ;; This hops via ssh:thror, then sudo:root on thror
                   ((member target '("root-thror" "rt")) "/ssh:thror|sudo:thror:/root")

                   ;; Girion (TrueNAS) - Admin/Docker Management
                   ;; Note: Using ssh-nas-tunnel based on our SSH config
                   ((member target '("girion" "g")) "/ssh:truenas_admin@ssh-nas-tunnel:/mnt/main-pool")
                   
                   ;; Fallback: acts like normal cd
                   (t target))))
        (eshell/cd path)
        (eshell/echo (format "Warped to %s" path))))
    (defun eshell/proxsh ()
      "Open the SOCKS proxy tunnel asynchronously (to not block the eshell buffer)"
      (start-process "proxy-tunnel" "*proxy-tunnel*" "ssh" "-N" "-D" "1080" "my-proxy-tunnel")
      (eshell/echo "my-proxy-tunnel (defined in nix config) opened in background"))
    :defer t)

    (defun nij/eshell-env-indicator ()
      "Return a string indicating active Nix/Python environments."
      ;; TODO: what about other environments?
      (let ((env-str ""))
        (when (getenv "IN_NIX_SHELL")
          (setq env-str (concat env-str "❄ ")))
        (when (getenv "VIRTUAL_ENV")
          (setq env-str (concat env-str "🐍 ")))
        (if (string-empty-p env-str)
            ""
          (propertize env-str 'face '(:foreground "#7dcfff"))))) ;tokyo night cyan
    (defun nij/eshell-git-branch ()
      "Return the current git branch natively using the built-in vc-git."
      (when (and (not (file-remote-p default-directory))
                 (vc-find-root default-directory ".git"))
        (require 'vc-git)
        (let ((branch (car (vc-git-branches))))
          (when branch
            (format "  %s" (propertize branch 'face '(:foreground "#bb9af7"))))))) ;tokyo night magenta
    (defun nij/eshell-prompt ()
      "Two-line prompt: [host] env directory branch, then the prompt char on a newline."
      (concat
       "\n"
       ;; a: remote host indicator
       (if-let* ((host (file-remote-p default-directory 'host)))
           (propertize (format "[%s] " host) 'face '(:foreground "#f7768e")) "")
       ;; b: env indicator (nix/direnv)
       (nij/eshell-env-indicator)
       ;; c: directory path
       (propertize (abbreviate-file-name (eshell/pwd)) 'face '(:foreground "#7aa2f7" :weight bold))
       ;; d: git branch
       (or (nij/eshell-git-branch) "")
       ;; e: root / non-root prompt character on the new line
       (propertize (if (= (user-uid) 0) "\n# " "\n❯ ") 'face '(:foreground "#9ece6a"))))
    :defer t
    :custom (eshell-prompt-function #'nij/eshell-prompt))
